log = open(OUT / "driver.log", "a")

  ("merge-lineage", "QuantFactory/ScaduTorrent1.1-8b-model_stock-GGUF", "59fc03079ddaf71a94ae4844c19780d1de40fbde", "kloodia/lora-8b-math", "2dcbc0215a7ebf72c9b3a692e5f5b556e9054f22"),

fps = {}
def fp_for(slug, rev):
    key = (slug, rev)
    if key in fps: return fps[key]
    path = OUT / (slug.replace("/", "--") + ".fp.json")
    if path.exists():
        from modeldna.fingerprint.extract import Fingerprint
        fps[key] = Fingerprint.from_dict(json.load(open(path))); return fps[key]
    t0 = time.time()
    src = HubSource(slug, revision=rev)
    fp = extract_fingerprint(src, mode="fast", model_id=slug, revision=rev)
    d = fp.to_dict(); d["bytes_read"] = getattr(src, "bytes_read", fp.bytes_read)
    path.write_text(json.dumps(d))
    say(f"fingerprinted {slug}@{rev[:8]} in {time.time()-t0:.0f}s, {d['bytes_read']/1e6:.0f} MB")
    fps[key] = fp; return fp

for scenario, s, srev, c, crev in PAIRS:
    name = f"{scenario}__{s}__vs__{c}".replace("/", "--") + ".json"
    out = OUT / name
    if out.exists(): continue
    rec = {"tool": "modeldna", "tool_version": __version__, "mode": "fast", "scenario": scenario,
           "subject": s, "subject_revision": srev, "candidate": c, "candidate_revision": crev}
    try:
        a = fp_for(s, srev); b = fp_for(c, crev)
        ev = compare_fingerprints(a, b)
        v = judge([(ev, c)], suspect_dtype=a.arch.torch_dtype, candidate_dtypes={c: b.arch.torch_dtype})
        rec["verdict"] = v.to_dict(); rec["evidence"] = ev.to_dict()
        say(f"{scenario}: {s} vs {c}: {v.verdict.value} p={v.probability}")
    except Exception as e:
        rec["error"] = f"{type(e).__name__}: {e}"; rec["traceback"] = traceback.format_exc()
        say(f"{scenario}: {s} vs {c}: NO VERDICT ({type(e).__name__}: {str(e)[:160]})")
    out.write_text(json.dumps(rec, indent=1))
say("done")
